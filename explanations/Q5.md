```asm
; Question 5
; Interactive string operations: length, comparison, and reversal.
; Self-contained: STRLEN, STRCMP, STRREV, and all keyboard/screen I/O
; live in this one file (no INCLUDE, no separate library).

STACKSEG SEGMENT PARA STACK 'STACK'
    DW 128 DUP (?)
STACKSEG ENDS

DATASEG SEGMENT PARA PUBLIC 'DATA'
    Q5_TITLE      DB 13,10,'QUESTION 5 - STRING OPERATIONS',13,10,'$'
    Q5_RANGE      DB 'Enter strings of up to 30 characters.',13,10,'$'
    FIRST_PROMPT  DB 'String 1: $'
    SECOND_PROMPT DB 'String 2: $'
    LEN1_LABEL    DB 'Length of string 1: $'
    LEN2_LABEL    DB 'Length of string 2: $'
    CMP_LABEL     DB 'Comparison: $'
    CMP_LESS      DB 'string 1 is less than string 2',13,10,'$'
    CMP_GREATER   DB 'string 1 is greater than string 2',13,10,'$'
    CMP_EQUAL     DB 'string 1 equals string 2',13,10,'$'
    REV_LABEL     DB 'Reversed string 1: $'

    ; DOS function 0AH buffer: maximum length, actual length, characters.
    ; Empirically verified against this authentic DOSBox 0.74-3 / MASM 5.10
    ; environment (see Q2.ASM/Q3.ASM/Q4.ASM): typing exactly maxlen
    ; characters never returns from INT 21H/AH=0AH here, so maxlen must sit
    ; well above the longest valid entry. Valid input is up to 30 characters,
    ; so maxlen=40 leaves generous headroom (well beyond the roughly-double
    ; margin used for Q4's much shorter fields); the char area is sized
    ; maxlen+1 (41) to also hold the CR DOS writes after the typed characters.
    STRING1_BUFFER DB 40,0,41 DUP (0)
    STRING2_BUFFER DB 40,0,41 DUP (0)

DATASEG ENDS

CODESEG SEGMENT PARA PUBLIC 'CODE'
    ASSUME CS:CODESEG, DS:DATASEG, SS:STACKSEG

START:
    MOV AX, DATASEG
    MOV DS, AX

    LEA DX, Q5_TITLE
    CALL PRINT_STRING
    LEA DX, Q5_RANGE
    CALL PRINT_STRING

    LEA DX, FIRST_PROMPT
    CALL PRINT_STRING
    LEA DX, STRING1_BUFFER
    CALL READ_STRING
    CALL PRINT_NEWLINE

    LEA DX, SECOND_PROMPT
    CALL PRINT_STRING
    LEA DX, STRING2_BUFFER
    CALL READ_STRING
    CALL PRINT_NEWLINE

    ; Length of string 1
    LEA SI, STRING1_BUFFER+2
    CALL STRLEN
    LEA DX, LEN1_LABEL
    CALL PRINT_STRING
    CALL PRINT_UINT
    CALL PRINT_NEWLINE

    ; Length of string 2
    LEA SI, STRING2_BUFFER+2
    CALL STRLEN
    LEA DX, LEN2_LABEL
    CALL PRINT_STRING
    CALL PRINT_UINT
    CALL PRINT_NEWLINE

    ; Comparison
    LEA SI, STRING1_BUFFER+2
    LEA DI, STRING2_BUFFER+2
    CALL STRCMP
    LEA DX, CMP_LABEL
    CALL PRINT_STRING
    CMP AX, 0FFFFH
    JE Q5_LESS
    CMP AX, 0
    JE Q5_EQUAL
    LEA DX, CMP_GREATER
    CALL PRINT_STRING
    JMP SHORT Q5_REVERSE
Q5_LESS:
    LEA DX, CMP_LESS
    CALL PRINT_STRING
    JMP SHORT Q5_REVERSE
Q5_EQUAL:
    LEA DX, CMP_EQUAL
    CALL PRINT_STRING

Q5_REVERSE:
    LEA SI, STRING1_BUFFER+2
    CALL STRREV
    LEA DX, REV_LABEL
    CALL PRINT_STRING
    LEA DX, STRING1_BUFFER+2
    CALL PRINT_Z
    CALL PRINT_NEWLINE

    MOV AX, 4C00H
    INT 21H

; Read a line into the DOS function 0AH buffer whose address is passed in
; DX, then null-terminate it by overwriting the trailing carriage return
; DOS leaves after the typed characters with a zero byte.
; Buffer layout: byte 0 = maxlen, byte 1 = actual typed length (set by
; DOS), bytes 2.. = typed characters. So the character data always starts
; at offset +2 regardless of maxlen; only that leading 2-byte header size
; is fixed, not the maxlen value itself.
; Input:  DX = buffer address (byte 0 already holds maxlen)
; Output: the string starting at [DX+2] is zero-terminated in place.
READ_STRING PROC NEAR
    PUSH AX
    PUSH BX
    PUSH SI

    MOV AH, 0AH
    INT 21H

    MOV SI, DX
    XOR BX, BX
    MOV BL, [SI+1]
    ADD SI, BX
    MOV BYTE PTR [SI+2], 0

    POP SI
    POP BX
    POP AX
    RET
READ_STRING ENDP

; Input:  DS:SI -> zero-terminated string
; Output: AX = character count (terminating zero is not counted)
; SI is preserved.
STRLEN PROC NEAR
    PUSH SI
    XOR AX, AX
SL_NEXT:
    CMP BYTE PTR [SI], 0
    JE SL_DONE
    INC SI
    INC AX
    JMP SHORT SL_NEXT
SL_DONE:
    POP SI
    RET
STRLEN ENDP

; Input:  DS:SI -> first string, DS:DI -> second string
; Output: AX = FFFFH if first < second, 0000H if equal, 0001H if first > second
; SI and DI are preserved.
STRCMP PROC NEAR
    PUSH BX
    PUSH SI
    PUSH DI
SC_NEXT:
    MOV AL, [SI]
    MOV BL, [DI]
    CMP AL, BL
    JB SC_LESS
    JA SC_GREATER
    OR AL, AL
    JZ SC_EQUAL
    INC SI
    INC DI
    JMP SHORT SC_NEXT
SC_LESS:
    MOV AX, 0FFFFH
    JMP SHORT SC_DONE
SC_GREATER:
    MOV AX, 1
    JMP SHORT SC_DONE
SC_EQUAL:
    XOR AX, AX
SC_DONE:
    POP DI
    POP SI
    POP BX
    RET
STRCMP ENDP

; Input:  DS:SI -> writable zero-terminated string
; Output: AX = character count; the string is reversed in place.
; SI is preserved.
STRREV PROC NEAR
    PUSH BX
    PUSH CX
    PUSH DX
    PUSH SI
    PUSH DI
    CALL STRLEN
    MOV DX, AX
    CMP AX, 1
    JBE SR_DONE
    MOV DI, SI
    ADD DI, AX
    DEC DI
    MOV CX, AX
    SHR CX, 1
SR_SWAP:
    MOV AL, [SI]
    MOV AH, [DI]
    MOV [SI], AH
    MOV [DI], AL
    INC SI
    DEC DI
    LOOP SR_SWAP
SR_DONE:
    MOV AX, DX
    POP DI
    POP SI
    POP DX
    POP CX
    POP BX
    RET
STRREV ENDP

; Print the dollar-terminated string at DS:DX.
PRINT_STRING PROC NEAR
    PUSH AX
    PUSH DX
    MOV AH, 09H
    INT 21H
    POP DX
    POP AX
    RET
PRINT_STRING ENDP

; Print the zero-terminated string at DS:DX by printing each character
; with DOS function 02H until a zero byte is reached. Needed because DOS
; function 09H requires a $-terminated string, but the user-entered and
; reversed strings in this program are zero-terminated.
PRINT_Z PROC NEAR
    PUSH AX
    PUSH DX
    PUSH SI
    MOV SI, DX
PZ_NEXT:
    MOV DL, [SI]
    CMP DL, 0
    JE PZ_DONE
    MOV AH, 02H
    INT 21H
    INC SI
    JMP SHORT PZ_NEXT
PZ_DONE:
    POP SI
    POP DX
    POP AX
    RET
PRINT_Z ENDP

; Print a carriage return and line feed.
PRINT_NEWLINE PROC NEAR
    PUSH AX
    PUSH DX
    MOV DL, 13
    MOV AH, 02H
    INT 21H
    MOV DL, 10
    MOV AH, 02H
    INT 21H
    POP DX
    POP AX
    RET
PRINT_NEWLINE ENDP

; Print the unsigned 16-bit value in AX as decimal.
PRINT_UINT PROC NEAR
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX

    MOV BX, 10
    XOR CX, CX
PU_DIVIDE:
    XOR DX, DX
    DIV BX
    PUSH DX
    INC CX
    OR AX, AX
    JNZ PU_DIVIDE
PU_DIGIT:
    POP DX
    ADD DL, '0'
    MOV AH, 02H
    INT 21H
    LOOP PU_DIGIT

    POP DX
    POP CX
    POP BX
    POP AX
    RET
PRINT_UINT ENDP

CODESEG ENDS
END START
```

Question 5 is a very useful jump because now you’re working with **strings as arrays of bytes** rather than just individual numbers. 🔥

The program does three main operations on two user-entered strings:

```text
1. Find the length of both strings
2. Compare the two strings lexicographically
3. Reverse string 1
```

The most important new ideas are:

```text
zero-terminated strings
pointer traversal
lexicographic comparison
two-pointer swapping
in-place reversal
```

---

# 1. First important idea: what is a string in assembly?

In C, you might write:

```c
char s[] = "HELLO";
```

Internally, memory contains:

```text
'H' 'E' 'L' 'L' 'O' 00H
```

That final:

```text
00H
```

marks the end of the string.

This is called a **null-terminated** or **zero-terminated string**.

So:

```text
HELLO
```

actually occupies:

```text
48H 45H 4CH 4CH 4FH 00H
```

The `00H` is not part of the visible string.

---

# 2. But DOS strings earlier used `$`

You have already seen things like:

```asm
Q5_TITLE DB 'QUESTION 5 - STRING OPERATIONS','$'
```

These are **DOS function 09H strings**.

They end using:

```text
$
```

rather than zero.

So this program deals with two kinds of strings:

```text
Program messages:
    '$'-terminated

User-entered strings:
    00H-terminated
```

That's why there are two different printing procedures:

```asm
PRINT_STRING
```

for `$` strings, and:

```asm
PRINT_Z
```

for zero-terminated strings.

This distinction is very important.

---

# 3. Input buffers

You have:

```asm
STRING1_BUFFER DB 40,0,41 DUP (0)
STRING2_BUFFER DB 40,0,41 DUP (0)
```

Remember DOS function `0AH` buffer layout:

```text
offset +0 → maximum input length
offset +1 → actual number of typed characters
offset +2 → first character
offset +3 → second character
...
```

So after typing:

```text
HELLO
```

the first buffer conceptually looks like:

```text
STRING1_BUFFER+0 = 40
STRING1_BUFFER+1 = 5

STRING1_BUFFER+2 = 'H'
STRING1_BUFFER+3 = 'E'
STRING1_BUFFER+4 = 'L'
STRING1_BUFFER+5 = 'L'
STRING1_BUFFER+6 = 'O'
```

DOS also leaves a carriage return after those characters.

The program replaces that carriage return with:

```text
00H
```

so that the input becomes a normal zero-terminated string.

---

# 4. Reading string 1

```asm
LEA DX, FIRST_PROMPT
CALL PRINT_STRING

LEA DX, STRING1_BUFFER
CALL READ_STRING
```

Notice carefully:

```asm
LEA DX, STRING1_BUFFER
```

not:

```asm
LEA DX, STRING1_BUFFER+2
```

Why?

Because DOS function `0AH` needs the **whole buffer**, including its first two control bytes.

So DX points here:

```text
        DX
         ↓
+0  maximum length
+1  actual length
+2  first character
...
```

---

# 5. But string operations use `+2`

Later:

```asm
LEA SI, STRING1_BUFFER+2
CALL STRLEN
```

Now we don't care about the DOS buffer header.

We want the actual characters.

So:

```text
STRING1_BUFFER     → DOS buffer
STRING1_BUFFER+2   → actual string
```

That's a key distinction.

---

# 6. READ_STRING

Let's examine:

```asm
READ_STRING PROC NEAR
    PUSH AX
    PUSH BX
    PUSH SI

    MOV AH, 0AH
    INT 21H
```

DX already contains the buffer address.

DOS reads the keyboard input into that buffer.

---

# 7. Find the end of the entered string

Next:

```asm
MOV SI, DX
```

So SI now points to the beginning of the DOS buffer.

Then:

```asm
XOR BX, BX
MOV BL, [SI+1]
```

`[SI+1]` contains the number of characters typed.

Suppose:

```text
HELLO
```

was entered.

Then:

```text
BL = 5
```

and because BX was zeroed:

```text
BX = 0005H
```

---

# 8. Move SI to the end

Then:

```asm
ADD SI, BX
```

Initially SI points at buffer offset 0.

After adding length:

```text
SI = buffer + length
```

For length 5:

```text
SI = buffer + 5
```

Then:

```asm
MOV BYTE PTR [SI+2], 0
```

So the actual location written is:

```text
buffer + 5 + 2
= buffer + 7
```

Why `+2`?

Because the actual characters begin at:

```text
buffer + 2
```

Therefore:

```text
buffer + 2 + length
```

is exactly the byte immediately after the last character.

So it writes:

```text
00H
```

there.

---

# 9. Example of null termination

Suppose the user enters:

```text
CAT
```

Length:

```text
3
```

DOS buffer:

```text
+0  40
+1  3
+2  'C'
+3  'A'
+4  'T'
+5  CR
```

The code writes:

```text
+5 = 00H
```

so now:

```text
'C' 'A' 'T' 00H
```

Perfect zero-terminated string.

---

# 10. `BYTE PTR`

This is a new syntax:

```asm
MOV BYTE PTR [SI+2], 0
```

Why say:

```text
BYTE PTR
```

?

Because:

```text
[SI+2]
```

is just a memory address.

MASM needs to know whether you want to store:

```text
8 bits?
16 bits?
```

The value `0` doesn't tell it clearly enough.

So:

```asm
BYTE PTR
```

explicitly says:

> treat this memory location as one byte.

You may also see:

```asm
WORD PTR
```

for a 16-bit memory value.

---

# 11. STRLEN

Now we reach the first actual string algorithm:

```asm
STRLEN PROC NEAR
    PUSH SI
    XOR AX, AX
```

Input:

```text
SI → first character
```

Output:

```text
AX = length
```

AX starts at:

```text
0
```

---

# 12. Traverse until zero

```asm
SL_NEXT:
    CMP BYTE PTR [SI], 0
    JE SL_DONE
```

This asks:

> Is the current character the terminating zero?

If yes:

```asm
JE SL_DONE
```

the string has ended.

Otherwise:

```asm
INC SI
INC AX
```

means:

```text
move to next character
length++
```

Then:

```asm
JMP SHORT SL_NEXT
```

repeats.

---

# 13. Example of STRLEN

Suppose memory contains:

```text
SI →
'H' 'E' 'L' 'L' 'O' 00
```

Initially:

```text
AX = 0
```

Iteration 1:

```text
'H' != 0
AX = 1
SI → 'E'
```

Iteration 2:

```text
AX = 2
```

then:

```text
AX = 3
AX = 4
AX = 5
```

finally:

```text
[SI] = 00H
```

so the procedure stops.

Result:

```text
AX = 5
```

---

# 14. Why `PUSH SI`?

The procedure modifies SI while scanning.

But its contract says:

```text
SI is preserved
```

So:

```asm
PUSH SI
```

saves the original pointer.

At the end:

```asm
POP SI
```

restores it.

Therefore after:

```asm
CALL STRLEN
```

the caller still has SI pointing at the start of the string.

This is good procedure design.

---

# 15. STRLEN in C

This assembly:

```asm
XOR AX, AX

SL_NEXT:
    CMP BYTE PTR [SI], 0
    JE SL_DONE
    INC SI
    INC AX
    JMP SL_NEXT
```

is approximately:

```c
length = 0;

while (s[length] != '\0') {
    length++;
}
```

Very straightforward.

---

# 16. Printing the lengths

For string 1:

```asm
LEA SI, STRING1_BUFFER+2
CALL STRLEN
```

After returning:

```text
AX = length
```

Then:

```asm
LEA DX, LEN1_LABEL
CALL PRINT_STRING
CALL PRINT_UINT
```

prints AX as decimal.

Important: `PRINT_STRING` preserves AX, which is why the length remains available for `PRINT_UINT`.

That's intentional.

---

# 17. String comparison

Now we reach the most interesting procedure:

```asm
STRCMP
```

The program loads:

```asm
LEA SI, STRING1_BUFFER+2
LEA DI, STRING2_BUFFER+2
CALL STRCMP
```

So:

```text
SI → string 1
DI → string 2
```

The procedure returns:

```text
AX = FFFFH → string1 < string2
AX = 0000H → equal
AX = 0001H → string1 > string2
```

---

# 18. What does "less than" mean for strings?

It means **lexicographic order**, like dictionary ordering.

For example:

```text
APPLE < BANANA
```

because:

```text
'A' < 'B'
```

ASCII values determine the comparison.

Another example:

```text
CAT < COW
```

Compare:

```text
C = C
A < O
```

so `CAT` is less.

We stop at the **first unequal character**.

---

# 19. STRCMP begins

```asm
STRCMP PROC NEAR
    PUSH BX
    PUSH SI
    PUSH DI
```

Again, registers used internally are preserved.

Then:

```asm
SC_NEXT:
    MOV AL, [SI]
    MOV BL, [DI]
```

Load corresponding characters.

For example:

```text
string1 = CAT
string2 = CAR
```

first iteration:

```text
AL = 'C'
BL = 'C'
```

---

# 20. Compare characters

```asm
CMP AL, BL
JB SC_LESS
JA SC_GREATER
```

Since ASCII byte values are treated as unsigned:

```text
AL < BL → string1 is smaller
AL > BL → string1 is greater
```

If neither jump occurs:

```text
AL == BL
```

so the characters are equal.

---

# 21. Why can one character decide the whole comparison?

Example:

```text
CAT
CAR
```

Positions:

```text
C = C
A = A
T > R
```

At the first difference:

```text
T > R
```

so immediately:

```text
CAT > CAR
```

There's no need to examine later characters.

This is exactly how C's `strcmp()` works conceptually.

---

# 22. What if the characters are equal?

Then:

```asm
OR AL, AL
JZ SC_EQUAL
```

This is checking whether:

```text
AL == 0
```

Why?

If:

```text
AL == BL
```

and both are zero:

```text
string1 ended
string2 ended
```

therefore both strings are identical.

Example:

```text
CAT\0
CAT\0
```

After comparing `C`, `A`, and `T`, eventually:

```text
AL = 0
BL = 0
```

So:

```asm
JZ SC_EQUAL
```

returns equality.

---

# 23. Why `OR AL,AL`?

You've seen this trick before.

```asm
OR AL, AL
```

does not change AL:

```text
x OR x = x
```

but updates the Zero Flag.

So:

```text
AL = 0 → ZF = 1
AL ≠ 0 → ZF = 0
```

Therefore:

```asm
JZ SC_EQUAL
```

means:

> If the equal characters we just compared are zero terminators, both strings have ended together.

---

# 24. Otherwise advance both pointers

```asm
INC SI
INC DI
JMP SHORT SC_NEXT
```

Move to the next character in both strings.

So comparison is:

```text
string1[0] vs string2[0]
string1[1] vs string2[1]
string1[2] vs string2[2]
...
```

---

# 25. What if one string is a prefix?

Consider:

```text
CAT
CATCH
```

Comparisons:

```text
C = C
A = A
T = T
```

Next:

```text
string1 character = 00H
string2 character = 'C'
```

So:

```text
00H < 'C'
```

and:

```asm
JB SC_LESS
```

returns:

```text
CAT < CATCH
```

which is correct dictionary-style ordering.

---

# 26. Return values

Less:

```asm
SC_LESS:
    MOV AX, 0FFFFH
```

Greater:

```asm
SC_GREATER:
    MOV AX, 1
```

Equal:

```asm
SC_EQUAL:
    XOR AX, AX
```

So the caller gets one of:

```text
FFFFH
0000H
0001H
```

You can mentally interpret `FFFFH` as `-1` in signed 16-bit representation, although the program just compares the raw value.

This mirrors typical `strcmp()` conventions:

```text
negative → less
zero     → equal
positive → greater
```

---

# 27. How the caller interprets STRCMP

After:

```asm
CALL STRCMP
```

you have:

```asm
CMP AX, 0FFFFH
JE Q5_LESS

CMP AX, 0
JE Q5_EQUAL
```

If neither matches, the only remaining result is:

```text
AX = 1
```

so string 1 is greater.

---

# 28. Case sensitivity

This comparison is **case-sensitive** because it directly compares ASCII values.

For example:

```text
'A' = 41H
'a' = 61H
```

So:

```text
'A' < 'a'
```

Therefore:

```text
APPLE < apple
```

according to this procedure.

That's not necessarily dictionary ordering humans expect, but it is standard raw ASCII lexicographic comparison.

---

# 29. Now STRREV: reversing a string

This is probably the most important new algorithm in Question 5.

Suppose:

```text
HELLO
```

We want:

```text
OLLEH
```

The procedure reverses the string **in place**.

That means it does not create another array.

It modifies the original memory directly.

---

# 30. Two-pointer idea

To reverse:

```text
H E L L O
```

swap:

```text
first ↔ last
```

giving:

```text
O E L L H
```

Then:

```text
second ↔ second-last
```

giving:

```text
O L L E H
```

Done.

So we use:

```text
SI → beginning
DI → end
```

and move them toward each other.

This is a classic two-pointer algorithm.

---

# 31. STRREV starts by finding length

```asm
CALL STRLEN
```

Remember:

```text
SI → string
```

and STRLEN returns:

```text
AX = length
```

SI itself is preserved by STRLEN.

Then:

```asm
MOV DX, AX
```

saves the length.

Why?

Because AX is going to be modified while reversing, but the procedure promises to return the original length in AX.

So:

```text
DX = saved length
```

---

# 32. Strings of length 0 or 1

```asm
CMP AX, 1
JBE SR_DONE
```

`JBE` means:

```text
Jump if Below or Equal
```

for unsigned comparison.

So:

```text
length <= 1
```

doesn't need reversing.

Examples:

```text
"" → unchanged

"A" → unchanged
```

---

# 33. Find the last character

```asm
MOV DI, SI
ADD DI, AX
DEC DI
```

This is important.

Suppose:

```text
SI = address of first character
AX = 5
```

for:

```text
HELLO
```

Then:

```asm
MOV DI, SI
ADD DI, AX
```

makes DI point to:

```text
SI + 5
```

But `SI+5` is the terminating zero:

```text
H E L L O 00
↑         ↑
SI       SI+5
```

We want the final actual character `'O'`.

Therefore:

```asm
DEC DI
```

makes:

```text
DI = SI + 4
```

which points to `'O'`.

Excellent pattern to remember:

```text
last element address = start + length - 1
```

---

# 34. How many swaps are needed?

```asm
MOV CX, AX
SHR CX, 1
```

`SHR` means:

```text
Shift Right
```

A right shift by one divides an unsigned number by 2.

So:

```text
CX = length / 2
```

Why only half?

Because every swap fixes **two characters**.

For length 6:

```text
ABCDEF
```

swaps:

```text
A ↔ F
B ↔ E
C ↔ D
```

Only:

```text
6 / 2 = 3
```

swaps.

For length 5:

```text
ABCDE
```

swaps:

```text
A ↔ E
B ↔ D
```

middle `C` stays where it is.

Integer division:

```text
5 / 2 = 2
```

exactly what we need.

---

# 35. `SHR CX,1`

Example:

```text
CX = 6 = 00000110
```

right shift:

```text
00000011 = 3
```

So:

```text
x >> 1 = x / 2
```

for unsigned integers.

This is the opposite idea of `SHL` from Question 3.

---

# 36. Swapping the characters

The core loop:

```asm
SR_SWAP:
    MOV AL, [SI]
    MOV AH, [DI]

    MOV [SI], AH
    MOV [DI], AL
```

Suppose:

```text
SI → 'H'
DI → 'O'
```

Then:

```text
AL = 'H'
AH = 'O'
```

and:

```asm
MOV [SI], AH
```

stores `'O'` at the front.

```asm
MOV [DI], AL
```

stores `'H'` at the end.

So:

```text
HELLO
```

becomes:

```text
OELLH
```

after the first swap.

---

# 37. Move inward

Then:

```asm
INC SI
DEC DI
```

So:

```text
SI → next character from left
DI → next character from right
```

Then:

```asm
LOOP SR_SWAP
```

repeats `length/2` times.

---

# 38. Worked reversal: `HELLO`

Initial:

```text
H E L L O
↑       ↑
SI      DI
```

Length:

```text
5
```

Number of swaps:

```text
5 / 2 = 2
```

First swap:

```text
O E L L H
  ↑   ↑
  SI  DI
```

Second swap:

```text
O L L E H
```

Result:

```text
OLLEH
```

Middle `L` never moves.

Perfect.

---

# 39. Return the length

At:

```asm
SR_DONE:
    MOV AX, DX
```

the original length saved in DX is restored to AX.

So the procedure both:

```text
reverses the string
```

and returns:

```text
AX = string length
```

even though the main program doesn't actually need that returned length afterward.

# 40. PRINT_Z

This is another important new helper procedure.

Why can't we do:

```asm
LEA DX, STRING1_BUFFER+2
CALL PRINT_STRING
```

?

Because `PRINT_STRING` uses:

```text
INT 21H / AH=09H
```

which expects:

```text
$-terminated strings
```

But string 1 is:

```text
zero-terminated
```

So we need a custom printing loop.

---

# 41. PRINT_Z logic

```asm
MOV SI, DX
```

DX contains the start address.

So:

```text
SI → first character
```

Then:

```asm
PZ_NEXT:
    MOV DL, [SI]
    CMP DL, 0
    JE PZ_DONE
```

Load one character and test:

```text
if character == 0:
    stop
```

Otherwise:

```asm
MOV AH, 02H
INT 21H
```

prints that one character.

Then:

```asm
INC SI
JMP SHORT PZ_NEXT
```

moves to the next character.

---

# 42. PRINT_Z in C-like pseudocode

It is essentially:

```c
while (*s != '\0') {
    putchar(*s);
    s++;
}
```

Very simple and very useful.

The entire Question 5 program is conceptually:

```c
read(string1);
read(string2);

len1 = strlen(string1);
len2 = strlen(string2);

print(len1);
print(len2);

cmp = strcmp(string1, string2);

if (cmp < 0)
    print("string 1 is less than string 2");
else if (cmp > 0)
    print("string 1 is greater than string 2");
else
    print("string 1 equals string 2");

reverse(string1);

```

That's exactly what the assembly implementation is doing.

# 47. 8085 connection

In 8085, you often scan memory using HL:

```asm
LXI H, ADDRESS

LOOP:
    MOV A, M
    ...
    INX H
    JMP LOOP
```

Here the same idea is:

```asm
MOV SI, address

LOOP:
    MOV AL, [SI]
    ...
    INC SI
    JMP LOOP
```

So mentally:

```text
8085 HL/M
≈
8086 SI/[SI]
```

This isn't a strict hardware equivalence, but it's a very useful learning analogy.

# 49. New concepts from Question 5

The important new concepts are:

| Concept | Meaning |
|---|---|
| null terminator | `00H` marks end of string |
| `BYTE PTR` | explicitly specify an 8-bit memory operand |
| `STRLEN` | scan until zero and count |
| `STRCMP` | compare corresponding characters |
| lexicographic order | first differing character decides |
| prefix comparison | shorter string is smaller when prefix matches |
| `STRREV` | reverse using two pointers |
| `SHR` | right shift, often divide by 2 |
| in-place operation | modify original memory directly |
| `PRINT_Z` | print zero-terminated string manually |

# 50. The three algorithms you should know

For this question, focus heavily on these three blocks.

### Length

```asm
XOR AX, AX

SL_NEXT:
    CMP BYTE PTR [SI], 0
    JE SL_DONE
    INC SI
    INC AX
    JMP SL_NEXT
```

Meaning:

```text
until '\0':
    move forward
    count++
```

### Comparison

```asm
MOV AL, [SI]
MOV BL, [DI]

CMP AL, BL
JB SC_LESS
JA SC_GREATER

OR AL, AL
JZ SC_EQUAL

INC SI
INC DI
```

Meaning:

```text
compare characters

different → decide immediately

same and zero → strings equal

otherwise → compare next characters
```

### Reversal

```asm
CALL STRLEN

MOV DI, SI
ADD DI, AX
DEC DI

MOV CX, AX
SHR CX, 1

SR_SWAP:
    MOV AL, [SI]
    MOV AH, [DI]
    MOV [SI], AH
    MOV [DI], AL
    INC SI
    DEC DI
    LOOP SR_SWAP
```

Meaning:

```text
left pointer  → start
right pointer → end

repeat length/2 times:
    swap left and right
    left++
    right--
```

Those are the real lessons of Question 5.
